\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 1}]
    \noindent
    SI এককে একটি মাধ্যমে সঞ্চালিত তরঙ্গের সরণ সমীকরণ $y = 0.2 \sin(200\pi t - 4\pi x)$ দ্বারা প্রকাশ করা হলো, যেখানে $x$ ও $y$ মিটারে এবং $t$ সেকেন্ডে পরিমাপ করা হয়। তরঙ্গটির সঞ্চালন বেগ এবং দোলনের পর্যায়কাল কত?
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    The displacement equation of a wave propagating in a medium in SI units is given by $y = 0.2 \sin(200\pi t - 4\pi x)$, where $x$ and $y$ are measured in meters and $t$ in seconds. Determine the propagation velocity and the time period of oscillation of the wave.
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $50\text{ m/s}, 0.01\text{ s}$
    \item[\textbf{B)}] $100\text{ m/s}, 0.02\text{ s}$
    \item[\textbf{C)}] $25\text{ m/s}, 0.01\text{ s}$
    \item[\textbf{D)}] $50\text{ m/s}, 0.02\text{ s}$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $50\text{ m/s}, 0.01\text{ s}$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত সমীকরণ:}
    \[ y = 0.2 \sin(200\pi t - 4\pi x) \]
    
    আদর্শ তরঙ্গ সমীকরণ $y = A \sin(\omega t - kx)$ এর সাথে তুলনা করে পাই:
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item কৌণিক কম্পাঙ্ক, $\omega = 200\pi\text{ rad/s}$
        \item তরঙ্গ সংখ্যা, $k = 4\pi\text{ rad/m}$
    \end{itemize}

    \vspace{0.6em}
    \noindent\textbf{গণনা:} \\
    (i) পর্যায়কাল ($T$):
    \[ \omega = 200\pi \implies \frac{2\pi}{T} = 200\pi \implies T = \frac{2\pi}{200\pi} = \mathbf{0.01\text{ s}} \]
    
    (ii) সঞ্চালন বেগ ($v$):
    \[ v = \frac{\omega}{k} = \frac{200\pi}{4\pi} = \mathbf{50\text{ m/s}} \]
\end{tcolorbox}

\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 2}]
    \noindent
    $500\text{ Hz}$ নির্দিষ্ট কম্পাঙ্কের একটি সুরশলাকা বাতাসে এবং পানিতে পৃথকভাবে অনুদৈর্ঘ্য তরঙ্গ সৃষ্টি করে। যদি বাতাসে ও পানিতে উৎপন্ন তরঙ্গদৈর্ঘ্য যথাক্রমে $0.68\text{ m}$ এবং $2.90\text{ m}$ পাওয়া যায়, তবে উক্ত মাধ্যম দুটিতে শব্দের সঞ্চালন বেগ যথাক্রমে কত হবে?
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    A tuning fork of a specific frequency $500\text{ Hz}$ produces longitudinal waves in air and water separately. If the wavelengths produced in air and water are found to be $0.68\text{ m}$ and $2.90\text{ m}$ respectively, what will be the propagation speeds of sound in these two media?
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $340\text{ m/s}, 1450\text{ m/s}$
    \item[\textbf{B)}] $330\text{ m/s}, 1400\text{ m/s}$
    \item[\textbf{C)}] $340\text{ m/s}, 1500\text{ m/s}$
    \item[\textbf{D)}] $350\text{ m/s}, 1450\text{ m/s}$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $340\text{ m/s}, 1450\text{ m/s}$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত উপাত্ত:}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item কম্পাঙ্ক, $f = 500\text{ Hz}$
        \item বাতাসে তরঙ্গদৈর্ঘ্য, $\lambda_{\text{air}} = 0.68\text{ m}$
        \item পানিতে তরঙ্গদৈর্ঘ্য, $\lambda_{\text{water}} = 2.90\text{ m}$
    \end{itemize}

    \vspace{0.6em}
    \noindent\textbf{গণনা:} \\
    (i) বাতাসে শব্দের বেগ:
    \[ v_{\text{air}} = f \lambda_{\text{air}} = 500 \times 0.68 = \mathbf{340\text{ m/s}} \]
    
    (ii) পানিতে শব্দের বেগ:
    \[ v_{\text{water}} = f \lambda_{\text{water}} = 500 \times 2.90 = \mathbf{1450\text{ m/s}} \]
\end{tcolorbox}

\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 3}]
    \noindent
    নির্দিষ্ট টানে রাখা $80\text{ cm}$ দৈর্ঘ্যের একটি টানা তারের মূল সুরের কম্পাঙ্ক $200\text{ Hz}$ পরিমাপ করা হলো। তারের ভর ও টান অপরিবর্তিত রেখে এর দৈর্ঘ্য কমিয়ে $64\text{ cm}$ করা হলে পরিবর্তিত অবস্থায় নতুন মূল সুরের কম্পাঙ্ক কত হবে?
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    The fundamental frequency of a stretched wire of length $80\text{ cm}$ under a constant tension is measured to be $200\text{ Hz}$. If the tension and mass of the wire remain unchanged and its length is reduced to $64\text{ cm}$, what will be the new fundamental frequency under the modified condition?
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $250\text{ Hz}$
    \item[\textbf{B)}] $225\text{ Hz}$
    \item[\textbf{C)}] $160\text{ Hz}$
    \item[\textbf{D)}] $300\text{ Hz}$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $250\text{ Hz}$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত উপাত্ত:}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item প্রাথমিক দৈর্ঘ্য, $L_1 = 80\text{ cm}$
        \item প্রাথমিক কম্পাঙ্ক, $f_1 = 200\text{ Hz}$
        \item পরিবর্তিত দৈর্ঘ্য, $L_2 = 64\text{ cm}$
    \end{itemize}

    \vspace{0.6em}
    \noindent\textbf{গণনা:} \\
    টানা তারের প্রথম সূত্রানুসারে, $f \propto \frac{1}{L}$ (যখন টান ও একক দৈর্ঘ্যের ভর স্থির থাকে)।
    \[ \frac{f_2}{f_1} = \frac{L_1}{L_2} \implies f_2 = f_1 \left(\frac{L_1}{L_2}\right) = 200 \times \left(\frac{80}{64}\right) = 200 \times 1.25 = \mathbf{250\text{ Hz}} \]
\end{tcolorbox}

\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 4}]
    \noindent
    এক ব্যক্তি খাড়া পাহাড় থেকে কিছুটা দূরে দাঁড়িয়ে স্থির অবস্থান থেকে বন্দুক থেকে একটি গুলি ছুড়ল এবং ঠিক $0.8\text{ s}$ পর শব্দের স্পষ্ট প্রতিধ্বনি শুনতে পেল। বায়ুমণ্ডলে শব্দের সঞ্চালন বেগ $340\text{ m/s}$ হলে, পাহাড় থেকে ব্যক্তির অবস্থান কত দূরে ছিল?
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    A person standing at a distance from a vertical cliff fires a gun from rest and hears a distinct echo of the sound after exactly $0.8\text{ s}$. If the speed of sound propagation in air is $340\text{ m/s}$, how far was the person located from the cliff?
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $136\text{ m}$
    \item[\textbf{B)}] $272\text{ m}$
    \item[\textbf{C)}] $170\text{ m}$
    \item[\textbf{D)}] $200\text{ m}$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $136\text{ m}$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত উপাত্ত:}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item শব্দের বেগ, $v = 340\text{ m/s}$
        \item প্রতিধ্বনি শোনার সময়, $t = 0.8\text{ s}$
    \end{itemize}

    \vspace{0.6em}

\end{tcolorbox}

\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 5}]
    \noindent
    $256\text{ Hz}$ কম্পাঙ্কের একটি জানা সুরশলাকা $B$ একটি অজানা সুরশলাকা $A$ এর সাথে একত্রে শব্দায়িত করা হলে প্রতি সেকেন্ডে $4$ টি বীট উৎপন্ন হয়। এরপর $A$ এর বাহুতে কিছুটা মোম লাগিয়ে ওজন বাড়ানো হলে বীট সংখ্যা কমে প্রতি সেকেন্ডে $2$ টি হয়। সুরশলাকা $A$ এর প্রাথমিক মূল কম্পাঙ্ক কত ছিল?
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    A known tuning fork $B$ of frequency $256\text{ Hz}$ sounded together with an unknown tuning fork $A$ produces $4$ beats per second. When some wax is attached to a prong of $A$ to increase its mass, the beat frequency decreases to $2$ beats per second. What was the original fundamental frequency of tuning fork $A$?
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $260\text{ Hz}$
    \item[\textbf{B)}] $252\text{ Hz}$
    \item[\textbf{C)}] $264\text{ Hz}$
    \item[\textbf{D)}] $248\text{ Hz}$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $260\text{ Hz}$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত উপাত্ত:}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item সুরশলাকা $B$ এর কম্পাঙ্ক, $f_B = 256\text{ Hz}$
        \item প্রাথমিক বীট সংখ্যা, $N = 4\text{ Hz}$
    \end{itemize}

    \vspace{0.6em}
    \noindent\textbf{বিশ্লেষণ ও গণনা:} \\
    সুরশলাকা $A$ এর সম্ভাব্য কম্পাঙ্ক:
    \[ f_A = f_B \pm N = 256 \pm 4 = 260\text{ Hz} \quad \text{বা} \quad 252\text{ Hz} \]
    
    সুরশলাকা $A$-তে মোম লাগালে এর কম্পাঙ্ক $f_A$ হ্রাস পায়। 
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item যদি $f_A = 260\text{ Hz}$ হয়, তবে $f_A$ কমে $258\text{ Hz}$ হলে বীট সংখ্যা $258 - 256 = 2$ হয় (যা প্রশ্নের সাথে মিলে)।
        \item যদি $f_A = 252\text{ Hz}$ হতো, তবে $f_A$ কমে $250\text{ Hz}$ হলে বীট সংখ্যা $256 - 250 = 6$ হতো (যা প্রশ্নের বিপরীত)।
    \end{itemize">
    
    অতএব, সুরশলাকা $A$ এর প্রাথমিক মূল কম্পাঙ্ক $f_A = \mathbf{260\text{ Hz}}$।
\end{tcolorbox}

\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 6}]
    \noindent
    $0.60\text{ m}$ দৈর্ঘ্যের একটি দুইমুখ খোলা বায়ুনলের মূল সুরের কম্পাঙ্ক এবং $0.30\text{ m}$ দৈর্ঘ্যের একটি একমুখ খোলা বায়ুনলের মূল সুরের কম্পাঙ্কের অনুপাত কত হবে, যখন উভয় নলের মাধ্যমেই শব্দের বেগ সমান?
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    What will be the ratio of the fundamental frequency of an open organ pipe of length $0.60\text{ m}$ to the fundamental frequency of a closed organ pipe of length $0.30\text{ m}$, given that the speed of sound is the same in both pipes?
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $1.0$
    \item[\textbf{B)}] $2.0$
    \item[\textbf{C)}] $0.5$
    \item[\textbf{D)}] $1.5$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $1.0$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত উপাত্ত:}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item খোলা বায়ুনলের দৈর্ঘ্য, $L_o = 0.60\text{ m}$
        \item বন্ধ বায়ুনলের দৈর্ঘ্য, $L_c = 0.30\text{ m}$
    \end{itemize}

    \vspace{0.6em}
    \noindent\textbf{গণনা:} \\
    (i) দুইমুখ খোলা বায়ুনলের মূল সুরের কম্পাঙ্ক:
    \[ f_o = \frac{v}{2 L_o} = \frac{v}{2(0.60)} = \frac{v}{1.20} \]
    
    (ii) একমুখ খোলা (বন্ধ) বায়ুনলের মূল সুরের কম্পাঙ্ক:
    \[ f_c = \frac{v}{4 L_c} = \frac{v}{4(0.30)} = \frac{v}{1.20} \]
    
    (iii) কম্পাঙ্কের অনুপাত:
    \[ \frac{f_o}{f_c} = \frac{v / 1.20}{v / 1.20} = \mathbf{1.0} \]
\end{tcolorbox}

    }

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]

\begin{tcolorbox}[questionbox={প্রশ্ন (Question) 25}]
    \noindent
    একটি শব্দ উৎস চারদিকে সুষমভাবে গোলকীয় তরঙ্গ আকারে শব্দ নির্গত করে। উৎস হতে $r_1 = 5\text{ m}$ দূরত্বে শব্দের তীব্রতা লেভেল $\beta_1 = 90\text{ dB}$। বায়ুর ঘনত্ব $\rho = 1.2\text{ kg/m}^3$ এবং শব্দের বেগ $v = 340\text{ m/s}$ হলে, উৎসের মোট ক্ষমতা ($P_{\text{source}}$), যে দূরত্বে তীব্রতা লেভেল কমে $70\text{ dB}$ হবে সেই দূরত্ব ($r_2$) এবং নতুন স্থানে শব্দের চাপ বিস্তার ($P_{\text{max},2}$) কত? ($I_0 = 10^{-12}\text{ W/m}^2$)
    
    \vspace{0.8em}
    \begin{english}
    \noindent\textit{\color{darkgray}
    A sound source emits spherical waves uniformly in all directions. At a distance $r_1 = 5\text{ m}$ from the source, the sound intensity level is $\beta_1 = 90\text{ dB}$. If the density of air is $\rho = 1.2\text{ kg/m}^3$ and speed of sound is $v = 340\text{ m/s}$, what are the total power of the source ($P_{\text{source}}$), the distance ($r_2$) where the intensity level drops to $70\text{ dB}$, and the sound pressure amplitude ($P_{\text{max},2}$) at the new location? ($I_0 = 10^{-12}\text{ W/m}^2$)
    }
    \end{english}
\end{tcolorbox}

\begin{itemize}[label={}, leftmargin=1cm, itemsep=0.5em]
    \item[\textbf{A)}] $0.1\pi\text{ W}, 50\text{ m}, 0.064\text{ N/m}^2$
    \item[\textbf{B)}] $0.2\pi\text{ W}, 100\text{ m}, 0.128\text{ N/m}^2$
    \item[\textbf{C)}] $0.1\pi\text{ W}, 25\text{ m}, 0.064\text{ N/m}^2$
    \item[\textbf{D)}] $0.05\pi\text{ W}, 50\text{ m}, 0.032\text{ N/m}^2$
\end{itemize}

\vspace{0.5em}
\begin{tcolorbox}[answerbox]
    \textbf{\color{success} উত্তর:} A) $0.1\pi\text{ W}, 50\text{ m}, 0.064\text{ N/m}^2$
\end{tcolorbox}
\vspace{1em}

\begin{tcolorbox}[solutionbox={সমাধান (Solution)}]
    \noindent
    \textbf{প্রদত্ত উপাত্ত:}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
        \item $r_1 = 5\text{ m}$, $\beta_1 = 90\text{ dB} \implies I_1 = 10^{-12} \times 10^9 = 10^{-3}\text{ W/m}^2$
        \item $\rho = 1.2\text{ kg/m}^3$, $v = 340\text{ m/s}$
    \end{itemize}

    \vspace{0.6em}
    \noindent\textbf{গণনা:} \\
    (i) উৎসের মোট ক্ষমতা:
    \[ P_{\text{source}} = I_1 \times 4\pi r_1^2 = 10^{-3} \times 4\pi (5)^2 = \mathbf{0.1\pi\text{ W}} \]
    
    (ii) $\beta_2 = 70\text{ dB} \implies I_2 = 10^{-5}\text{ W/m}^2$:
    \[ \frac{I_1}{I_2} = \left(\frac{r_2}{r_1}\right)^2 \implies \frac{10^{-3}}{10^{-5}} = \left(\frac{r_2}{5}\right)^2 \implies 100 = \left(\frac{r_2}{5}\right)^2 \implies r_2 = \mathbf{50\text{ m}} \]
    
    (iii) নতুন স্থানে শব্দের চাপ বিস্তার:
    \[ P_{\text{max},2} = \sqrt{I_2 \rho v} = \sqrt{10^{-5} \times 1.2 \times 340} = \sqrt{0.00408} \approx \mathbf{0.064\text{ N/m}^2} \]
\end{tcolorbox}
